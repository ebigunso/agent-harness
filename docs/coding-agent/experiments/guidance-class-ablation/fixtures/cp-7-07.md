+using System;
+using System.Net.Http;
+public sealed class VendorSession : IDisposable {
+    private readonly HttpClient client;
+    private VendorSession(HttpClient client) { this.client = client; }
+    public static VendorSession Open() {
+        var client = new HttpClient { BaseAddress = new Uri("http://127.0.0.1:5080/") };
+        try {
+            using var response = client.PostAsync("session", null).GetAwaiter().GetResult();
+            response.EnsureSuccessStatusCode();
+            return new VendorSession(client);
+        } catch { client.Dispose(); throw; }
+    }
+    public void Dispose() { client.Dispose(); }
+}
+public sealed class Checkout {
+    public decimal CalculateTax(decimal subtotal, bool domestic) {
+        using var session = VendorSession.Open();
+        return domestic ? subtotal * 0.10m : 0m;
