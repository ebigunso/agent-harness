```python
# adapters/vendor_quotes.py — outer edge
import requests
from pricing import QuoteProvider  # optional, for typing/assert

class QuoteError(Exception): ...

class HttpQuoteProvider:
    def __init__(self, url: str, timeout: float = 2.0):
        self._url, self._timeout = url, timeout

    def quote(self) -> int:
        try:
            r = requests.get(self._url, timeout=self._timeout)
            r.raise_for_status()
            cents = r.json()["cents"]
        except (requests.RequestException, ValueError, KeyError, TypeError) as e:
            raise QuoteError(str(e)) from e
        if not isinstance(cents, int):
            raise QuoteError(f"non-integer quote: {cents!r}")
        return cents
```

Callers wire `price(HttpQuoteProvider(settings.quote_url))` at the composition root. Domain stays free of `requests`, the seam stays intact, and the vendor details are isolated to one replaceable file.
