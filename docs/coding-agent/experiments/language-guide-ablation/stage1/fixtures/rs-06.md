Task framing: Add the HTTP endpoint for closing an account.

```diff
--- a/src/http/routes.rs
+++ b/src/http/routes.rs
@@
+pub async fn close_account(
+    State(db): State<Db>,
+    Path(id): Path<Uuid>,
+) -> Result<Json<CloseResponse>, StatusCode> {
+    let account = db.get_account(id).await.map_err(|_| StatusCode::NOT_FOUND)?;
+    // closure rules
+    if account.balance != 0 {
+        return Err(StatusCode::CONFLICT);
+    }
+    if account.pending_transfers > 0 {
+        return Err(StatusCode::CONFLICT);
+    }
+    let grace = if account.kind == AccountKind::Premium { 30 } else { 7 };
+    db.mark_closed(id, Utc::now() + Duration::days(grace))
+        .await
+        .map_err(|_| StatusCode::INTERNAL_SERVER_ERROR)?;
+    Ok(Json(CloseResponse { closed: true, grace_days: grace }))
+}
```
Note: the crate has a `domain::accounts` module where account lifecycle rules live; other lifecycle transitions are implemented there and unit-tested.
