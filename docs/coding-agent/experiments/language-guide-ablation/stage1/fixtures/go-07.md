--- /dev/null

+package invoice
+
+// Formatter formats invoice numbers for display.
+type Formatter interface {
+	Format(n Number) string
+}
+
+// NewFormatter returns the default Formatter.
+func NewFormatter(prefix string) Formatter {
+	return &formatter{prefix: prefix}
+}
+
+type formatter struct {
+	prefix string
+}
+
+func (f *formatter) Format(n Number) string {
+	return fmt.Sprintf("%s-%08d", f.prefix, n)
+}
