Task framing: The order-total calculation is needed by both the API and the batch exporter; make it reusable.

```diff
--- a/internal/util/helpers.go

@@
+// OrderTotal computes the payable total for an order including promo and zone fees.
+func OrderTotal(o *orders.Order, promos *billing.PromoTable) (money.Amount, error) {
+	base := o.Subtotal()
+	disc, err := promos.DiscountFor(o)
+	if err != nil {
+		return money.Zero, fmt.Errorf("order total: %w", err)
+	}
+	return base.Sub(disc).Add(shipping.ZoneFee(o.Destination)), nil
+}
```
